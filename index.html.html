        }

        function renderMessages() {

            const container = document.getElementById('chat-messages');

            const choicesContainer = document.getElementById('chat-choices');

            container.innerHTML = '';

            choicesContainer.innerHTML = '';

            const data = chatData[currentRoom];

           

            data.messages.forEach(msg => {

                const msgEl = document.createElement('div');

                msgEl.className = `msg ${msg.type === 'right' ? 'msg-right' : 'msg-left'}`;

               

                if(msg.type === 'left') {

                    msgEl.innerHTML = `<div class="msg-sender">${msg.sender}</div>${msg.text}`;

                } else {

                    msgEl.innerText = msg.text;

                }

               

                container.appendChild(msgEl);

            });

            container.scrollTop = container.scrollHeight;
